# One test definition for both the portable and ROS builds.
function(add_core_test name)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${name}.cpp")
    message(STATUS "Skipping unavailable test source: tests/${name}.cpp")
    return()
  endif()
  add_executable(${name} tests/${name}.cpp)
  target_include_directories(${name} PRIVATE include)
  target_compile_options(${name} PRIVATE -UNDEBUG)
  add_test(NAME ${name} COMMAND ${name} ${ARGN})
endfunction()
function(add_core_tests)
  foreach(test coordination_components_test incremental_repair_v19_test
      follower_alignment_test route_switch_test ultrasonic_memory_test
      v2_passage_handshake_test v210_entrance_test v210_route_channel_test
      v210_header_integration_test)
    add_core_test(${test})
  endforeach()
  add_core_test(map_snapshot_test)
  add_core_test(map_input_test)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  foreach(test v2_node_passage_test v210_launch_test v210_node_continuity_test v212_map_pipeline_test)
    add_test(NAME ${test} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/${test}.py)
  endforeach()
endfunction()
